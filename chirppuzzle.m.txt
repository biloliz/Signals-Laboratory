function y = mychirp(t, f0, f1)
  %mychirp Generate a chirp signal.
  %   y = mychirp(t, f0, f1) generates a chirp signal with
  %   instantaneous frequency that varies linearly with time.
  %
  %   t  : Time vector.
  %   f0 : Starting frequency (Hz).
  %   f1 : Ending frequency (Hz).
  %
  %   The instantaneous frequency at time t is given by:
  %   f(t) = f0 + (f1 - f0) * t / t(end)
  %
  %   The phase is the integral of 2*pi*f(t)

  T = t(end);  % Total duration
  k = (f1 - f0) / T; % Frequency slope
  phase = 2*pi * (f0*t + 0.5*k*t.^2);
  y = cos(phase);
end

% Parameters for the second chirp signal
fs = 11025;  % Sampling rate (Hz)
Tdur = 3;    % Total time duration (s)
f0 = 3000; % Starting frequency (Hz)
f1 = -2000; % Ending frequency (Hz)  <--  Key change: negative frequency

% Generate the time vector
dt = 1/fs;
t = 0:dt:Tdur;

% Generate the chirp signal using mychirp
chirpSignal2 = mychirp(t, f0, f1);


% Create and plot the spectrogram
figure;
spectrogram(chirpSignal2, 1024, 200, [], fs, 'yaxis');
colormap(flipud(gray(256)));
title('Spectrogram of Chirp Signal (3000 Hz to -2000 Hz)');
colorbar;
ylabel('Frequency (Hz)');
xlabel('Time (s)');

% Theoretical Explanation (Spectrum and Negative Frequencies)
% A cosine wave, like the one generated by mychirp, has a spectrum
% with components at both positive and negative frequencies.
%
% For a cosine wave cos(2*pi*f*t), there are two impulses in the
% frequency domain: one at +f and one at -f.  These are mathematical
% constructs.
%
% When the instantaneous frequency changes from 3000 Hz to -2000 Hz,
% the following happens in the spectrum:
%
% 1. The positive frequency component starts at +3000 Hz.
% 2. It moves downwards towards 0 Hz.
% 3. It continues to move towards -2000 Hz.
%
% The negative frequency component does the opposite
%
% 1. The negative frequency component starts at -3000 Hz.
% 2. It moves upwards towards 0 Hz.
% 3. It continues to move upwards towards +2000 Hz.
%
% Because our ears can only perceive positive frequencies, we hear
% the positive frequency component.  Therefore, we hear a tone that
% starts at 3000 Hz and decreases to 2000 Hz.
%
% The spectrogram displays the magnitude of the frequency components.
% It will show the positive frequency component moving from 3000 Hz
% down to 2000 Hz.  The negative frequencies are present in the
% mathematical representation of the signal, but are not directly
% visible in a standard spectrogram.
