% (a) Generate and Analyze the Hamming BPF (L=41)
L = 41;
omega_o = 0.25 * pi;
n = 0:L-1;
hamming_window = 0.54 - 0.46 * cos(2 * pi * n / (L - 1));
impulse_response = hamming_window .* cos(omega_o * (n - (L - 1) / 2));

% Calculate the frequency response
[h, w] = freqz(impulse_response, 1, 512);
magnitude = abs(h);
phase = angle(h);

% Plot the frequency response
figure;
subplot(2, 1, 1);
plot(w / pi, 20 * log10(magnitude));
title('Magnitude Response (L=41)');
xlabel('Frequency (\omega / \pi)');
ylabel('Magnitude (dB)');
grid on;

subplot(2, 1, 2);
plot(w / pi, phase);
title('Phase Response (L=41)');
xlabel('Frequency (\omega / \pi)');
ylabel('Phase (radians)');
grid on;

% Measure the response at specific frequencies
frequencies_oi = [0, 0.1 * pi, 0.25 * pi, 0.4 * pi, 0.5 * pi, 0.75 * pi];
response_table = zeros(length(frequencies_oi), 3);

for i = 1:length(frequencies_oi)
    omega = frequencies_oi(i);
    [~, index] = min(abs(w - omega));
    response_table(i, 1) = omega;
    response_table(i, 2) = magnitude(index);
    response_table(i, 3) = phase(index);
end

disp('Response at Frequencies of Interest (L=41):');
disp('--------------------------------------------');
disp('Frequency (rad/sample) | Magnitude | Phase (radians)');
disp('--------------------------------------------');
fprintf('%f              | %f    | %f\n', response_table');
disp('--------------------------------------------');

% Zooming to estimate stopband ripple height
figure;
plot(w / pi, 20 * log10(magnitude));
title('Magnitude Response (L=41) - Zoomed Stopband');
xlabel('Frequency (\omega / \pi)');
ylabel('Magnitude (dB)');
xlim([0, 0.05]); % Adjust xlim to focus on a stopband region
grid on;

% (b) Passband Width and Filter Length
% Function to calculate passband width
function passband_width = calculate_passband_width(magnitude, w)
    peak_magnitude = max(magnitude);
    passband_indices = find(magnitude >= 0.5 * peak_magnitude);
    if ~isempty(passband_indices)
        passband_start_index = passband_indices(1);
        passband_end_index = passband_indices(end);
        passband_width = (w(passband_end_index) - w(passband_start_index)) / pi;
    else
        passband_width = 0;
    end
end

% Analyze for L = 41 (already done)
passband_width_41 = calculate_passband_width(magnitude, w);
fprintf('\nPassband Width (L=41): %f pi radians/sample\n', passband_width_41);
